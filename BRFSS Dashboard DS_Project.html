# core aggregator (slightly generalized)
aggregate_question <- function(df, question_pattern, breakout_value = "Overall") {
  df %>%
    mutate(Data_value = as.numeric(Data_value),
           Sample_Size = as.numeric(Sample_Size)) %>%
    filter(str_detect(Question, regex(question_pattern, ignore_case=TRUE)),
           Break_Out == breakout_value,
           !is.na(Data_value), !is.na(Sample_Size)) %>%
    rename(persons = Sample_Size) %>%
    mutate(Sample_Size = persons * 100 / Data_value) %>%
    group_by(Response) %>%
    summarise(
      agg_ss = sum(Sample_Size, na.rm=TRUE),
      agg_persons = sum(persons, na.rm=TRUE),
      agg_percent = agg_persons * 100 / agg_ss,
      agg_percent_sdev = sqrt(agg_percent * (100 - agg_percent) / agg_ss),
      low_CI = agg_percent - 2 * agg_percent_sdev,
      high_CI = agg_percent + 2 * agg_percent_sdev
    ) %>%
    arrange(desc(agg_percent))
}

# plotting helper
plot_confidence_bar <- function(tbl, title=NULL) {
  plot_ly(tbl, x=~Response, y=~agg_percent, type="bar",
          error_y = list(type="data",
                         array = tbl$high_CI - tbl$agg_percent,
                         arrayminus = tbl$agg_percent - tbl$low_CI)) %>%
    layout(title=title, yaxis=list(title="Percentage (%)"))
}

# server
server <- function(input, output, session) {

  # topic UI depends on input$class
  output$topic_ui <- renderUI({
    req(input$class)                 # wait until class selected
    topics <- df %>% filter(Class == input$class) %>% pull(Topic) %>% unique()
    selectInput("topic", "Topic", choices = topics)
  })

  # question UI depends on input$topic
  output$question_ui <- renderUI({
    req(input$class, input$topic)
    qs <- df %>% filter(Class == input$class, Topic == input$topic) %>% pull(Question) %>% unique()
    selectInput("question", "Question", choices = qs)
  })

  # Example reactive that shows current selection (debug)
  output$debug <- renderPrint({
    list(class = input$class, topic = input$topic, question = input$question)
  })

  # Example: render a plot when a question is chosen
  output$overall_plot <- renderPlotly({
    req(input$question)
    tbl <- aggregate_question(df, question_pattern = input$question, breakout_value = "Overall")
    plot_confidence_bar(tbl, title = input$question)
  })
}
